//Experiment 25: UNIX I/O System Calls

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <dirent.h>

int main()
{
    int fd;
    struct stat st;
    DIR *dir;
    struct dirent *entry;

    fd = open("file.txt", O_CREAT | O_RDWR, 0644);

    printf("File opened using open()\n");

    fcntl(fd, F_GETFL);
    printf("fcntl() executed\n");

    lseek(fd, 0, SEEK_END);
    printf("lseek() executed\n");

    stat("file.txt", &st);
    printf("File Size: %ld bytes\n", st.st_size);

    dir = opendir(".");

    printf("\nDirectory Files:\n");

    while((entry = readdir(dir)) != NULL)
        printf("%s\n", entry->d_name);

    close(fd);
    closedir(dir);

    return 0;
}
