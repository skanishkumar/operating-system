//Experiment 36: Linked File Allocation

#include <stdio.h>

struct Block
{
    int block;
    int next;
};

int main()
{
    struct Block b[20];
    int n, i;

    printf("Enter number of blocks: ");
    scanf("%d", &n);

    printf("Enter block numbers:\n");

    for(i = 0; i < n; i++)
        scanf("%d", &b[i].block);

    for(i = 0; i < n - 1; i++)
        b[i].next = b[i + 1].block;

    b[n - 1].next = -1;

    printf("\nLinked Allocation:\n");

    for(i = 0; i < n; i++)
        printf("Block %d -> %d\n", b[i].block, b[i].next);

    return 0;
}
