//Experiment 10: Inter-Process Communication using Message Queue

#include <stdio.h>
#include <string.h>
#include <sys/ipc.h>
#include <sys/msg.h>

struct message
{
    long type;
    char text[100];
};

int main()
{
    key_t key = 1234;
    int msgid;

    struct message msg;

    // Create message queue
    msgid = msgget(key, 0666 | IPC_CREAT);

    msg.type = 1;
    strcpy(msg.text, "Hello using Message Queue");

    // Send message
    msgsnd(msgid, &msg, sizeof(msg.text), 0);

    printf("Message Sent: %s\n", msg.text);

    // Receive message
    msgrcv(msgid, &msg, sizeof(msg.text), 1, 0);

    printf("Message Received: %s\n", msg.text);

    // Remove queue
    msgctl(msgid, IPC_RMID, NULL);

    return 0;
}
