//Experiment 32: LRU Page Replacement

#include <stdio.h>

int main()
{
    int pages[50], frame[10], recent[10];
    int n, f, i, j, k, pos, fault = 0;
    int found, min;

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter page reference string:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &pages[i]);

    printf("Enter number of frames: ");
    scanf("%d", &f);

    for(i = 0; i < f; i++)
    {
        frame[i] = -1;
        recent[i] = 0;
    }

    for(i = 0; i < n; i++)
    {
        found = 0;

        for(j = 0; j < f; j++)
        {
            if(frame[j] == pages[i])
            {
                found = 1;
                recent[j] = i;
                break;
            }
        }

        if(!found)
        {
            pos = -1;

            for(j = 0; j < f; j++)
            {
                if(frame[j] == -1)
                {
                    pos = j;
                    break;
                }
            }

            if(pos == -1)
            {
                min = recent[0];
                pos = 0;

                for(k = 1; k < f; k++)
                {
                    if(recent[k] < min)
                    {
                        min = recent[k];
                        pos = k;
                    }
                }
            }

            frame[pos] = pages[i];
            recent[pos] = i;
            fault++;
        }

        printf("Page %d: ", pages[i]);

        for(j = 0; j < f; j++)
            printf("%d ", frame[j]);

        printf("\n");
    }

    printf("Total Page Faults = %d\n", fault);

    return 0;
}
