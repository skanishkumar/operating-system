//Experiment 24: UNIX System Calls for File Management

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main()
{
    int fd;
    char data[100];

    fd = open("file.txt", O_CREAT | O_RDWR, 0644);

    if(fd == -1)
    {
        printf("File opening failed\n");
        return 1;
    }

    write(fd, "Hello UNIX\n", 11);

    lseek(fd, 0, SEEK_SET);

    int n = read(fd, data, sizeof(data) - 1);
    data[n] = '\0';

    printf("File Content:\n%s", data);

    close(fd);

    return 0;
}
