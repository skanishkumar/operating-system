//Experiment 22: Best Fit Memory Allocation

#include <stdio.h>

int main()
{
    int block[20], process[20];
    int n, m, i, j, index;

    printf("Enter number of blocks: ");
    scanf("%d", &n);

    printf("Enter block sizes:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &block[i]);

    printf("Enter number of processes: ");
    scanf("%d", &m);

    printf("Enter process sizes:\n");
    for(i = 0; i < m; i++)
        scanf("%d", &process[i]);

    for(i = 0; i < m; i++)
    {
        index = -1;

        for(j = 0; j < n; j++)
        {
            if(block[j] >= process[i])
            {
                if(index == -1 || block[j] < block[index])
                    index = j;
            }
        }

        if(index != -1)
        {
            printf("P%d -> Block %d\n", i + 1, index + 1);
            block[index] -= process[i];
        }
        else
        {
            printf("P%d -> Not Allocated\n", i + 1);
        }
    }

    return 0;
}
