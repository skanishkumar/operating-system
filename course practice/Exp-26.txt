//Experiment 26: File Management Operations

#include <stdio.h>

int main()
{
    FILE *fp;
    char data[100];

    fp = fopen("sample.txt", "w");

    fprintf(fp, "Hello File Management\n");
    fclose(fp);

    fp = fopen("sample.txt", "r");

    fgets(data, sizeof(data), fp);

    printf("File Content: %s", data);

    fclose(fp);

    printf("\nFile operations completed successfully.\n");

    return 0;
}
