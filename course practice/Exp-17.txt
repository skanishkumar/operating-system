//Experiment 17: Banker's Algorithm

#include <stdio.h>

int main()
{
    int n, r;
    int alloc[10][10], max[10][10], need[10][10];
    int avail[10], finish[10] = {0}, safe[10];
    int i, j, k, count = 0, found;

    printf("Enter processes and resources: ");
    scanf("%d%d", &n, &r);

    printf("Enter Allocation matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < r; j++)
            scanf("%d", &alloc[i][j]);

    printf("Enter Maximum matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < r; j++)
            scanf("%d", &max[i][j]);

    printf("Enter Available resources:\n");
    for(j = 0; j < r; j++)
        scanf("%d", &avail[j]);

    for(i = 0; i < n; i++)
        for(j = 0; j < r; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    while(count < n)
    {
        found = 0;

        for(i = 0; i < n; i++)
        {
            if(!finish[i])
            {
                for(j = 0; j < r; j++)
                    if(need[i][j] > avail[j])
                        break;

                if(j == r)
                {
                    for(k = 0; k < r; k++)
                        avail[k] += alloc[i][k];

                    safe[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(!found)
            break;
    }

    if(count == n)
    {
        printf("Safe Sequence: ");

        for(i = 0; i < n; i++)
            printf("P%d ", safe[i]);

        printf("\nSystem is Safe\n");
    }
    else
        printf("System is Unsafe\n");

    return 0;
}
