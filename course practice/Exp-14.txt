//Experiment 14: Single Level Directory

#include <stdio.h>
#include <string.h>

int main()
{
    char files[20][30];
    int n = 0, choice;
    char name[30];

    while(1)
    {
        printf("\n1.Create  2.Display  3.Search  4.Exit\n");
        scanf("%d", &choice);

        if(choice == 1)
        {
            printf("Enter file name: ");
            scanf("%s", name);

            strcpy(files[n++], name);
            printf("File created\n");
        }
        else if(choice == 2)
        {
            printf("Files:\n");

            for(int i = 0; i < n; i++)
                printf("%s\n", files[i]);
        }
        else if(choice == 3)
        {
            printf("Enter file name: ");
            scanf("%s", name);

            for(int i = 0; i < n; i++)
                if(strcmp(files[i], name) == 0)
                {
                    printf("File found\n");
                    return 0;
                }

            printf("File not found\n");
        }
        else
            break;
    }

    return 0;
}
