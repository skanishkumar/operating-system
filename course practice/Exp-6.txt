//Experiment 6: Preemptive Priority Scheduling (Simple)

#include <stdio.h>
#include <limits.h>

int main()
{
    int n, i, time = 0, completed = 0;
    int bt[20], rt[20], pr[20];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nProcess %d\n", i + 1);

        printf("Burst Time: ");
        scanf("%d", &bt[i]);

        printf("Priority: ");
        scanf("%d", &pr[i]);

        rt[i] = bt[i];
    }

    while(completed < n)
    {
        int highest = -1;
        int priority = INT_MAX;

        for(i = 0; i < n; i++)
        {
            if(rt[i] > 0 && pr[i] < priority)
            {
                priority = pr[i];
                highest = i;
            }
        }

        rt[highest]--;
        time++;

        if(rt[highest] == 0)
            completed++;
    }

    printf("\nAll processes executed successfully.\n");

    return 0;
}
