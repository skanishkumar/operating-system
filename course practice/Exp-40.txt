//Experiment 40: Linux File Access Permissions

#include <stdio.h>
#include <sys/stat.h>

int main()
{
    struct stat file;

    if(stat("file.txt", &file) != 0)
    {
        printf("File not found\n");
        return 1;
    }

    printf("File Permissions:\n");

    printf("Owner: ");

    if(file.st_mode & S_IRUSR)
        printf("Read ");

    if(file.st_mode & S_IWUSR)
        printf("Write ");

    if(file.st_mode & S_IXUSR)
        printf("Execute ");

    printf("\nGroup: ");

    if(file.st_mode & S_IRGRP)
        printf("Read ");

    if(file.st_mode & S_IWGRP)
        printf("Write ");

    if(file.st_mode & S_IXGRP)
        printf("Execute ");

    printf("\nOthers: ");

    if(file.st_mode & S_IROTH)
        printf("Read ");

    if(file.st_mode & S_IWOTH)
        printf("Write ");

    if(file.st_mode & S_IXOTH)
        printf("Execute ");

    printf("\n");

    return 0;
}
