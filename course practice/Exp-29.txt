//Experiment 29: Classical Process Synchronization

#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

sem_t semaphore;

void *process(void *arg)
{
    int id = *(int *)arg;

    sem_wait(&semaphore);

    printf("Process %d entered critical section\n", id);
    printf("Process %d is executing\n", id);
    printf("Process %d left critical section\n", id);

    sem_post(&semaphore);

    return NULL;
}

int main()
{
    pthread_t t1, t2;
    int a = 1, b = 2;

    sem_init(&semaphore, 0, 1);

    pthread_create(&t1, NULL, process, &a);
    pthread_create(&t2, NULL, process, &b);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    sem_destroy(&semaphore);

    return 0;
}
