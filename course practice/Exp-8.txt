//Experiment 8: Round Robin Scheduling

#include <stdio.h>

int main()
{
    int n, i, tq;
    int bt[20], rt[20];
    int time = 0, completed = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Burst Time P%d: ", i + 1);
        scanf("%d", &bt[i]);
        rt[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    while(completed < n)
    {
        for(i = 0; i < n; i++)
        {
            if(rt[i] > 0)
            {
                if(rt[i] <= tq)
                {
                    time += rt[i];
                    rt[i] = 0;
                    completed++;
                }
                else
                {
                    rt[i] -= tq;
                    time += tq;
                }
            }
        }
    }

    printf("\nRound Robin Scheduling Completed Successfully.\n");

    return 0;
}
