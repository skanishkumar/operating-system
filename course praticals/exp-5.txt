#include <stdio.h>

int main()
{
    int n, i, j, temp;
    int bt[20], pr[20], wt[20], tat[20];

    printf("Enter Number of Processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Burst Time and Priority for P%d: ", i + 1);
        scanf("%d%d", &bt[i], &pr[i]);
    }

    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(pr[i] > pr[j])
            {
                temp = pr[i];
                pr[i] = pr[j];
                pr[j] = temp;

                temp = bt[i];
                bt[i] = bt[j];
                bt[j] = temp;
            }
        }
    }

    wt[0] = 0;

    for(i = 1; i < n; i++)
        wt[i] = wt[i - 1] + bt[i - 1];

    for(i = 0; i < n; i++)
    {
        tat[i] = wt[i] + bt[i];
        printf("P%d\tBT=%d\tPR=%d\tWT=%d\tTAT=%d\n",
               i + 1, bt[i], pr[i], wt[i], tat[i]);
    }

    return 0;
}
