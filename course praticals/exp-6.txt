#include <stdio.h>

int main()
{
    int n, i, time = 0, completed = 0;
    int bt[20], rt[20], pr[20];
    int highest, minPriority;

    printf("Enter Number of Processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Burst Time and Priority P%d: ", i + 1);
        scanf("%d%d", &bt[i], &pr[i]);
        rt[i] = bt[i];
    }

    while(completed < n)
    {
        highest = -1;
        minPriority = 9999;

        for(i = 0; i < n; i++)
        {
            if(rt[i] > 0 && pr[i] < minPriority)
            {
                minPriority = pr[i];
                highest = i;
            }
        }

        rt[highest]--;
        time++;

        if(rt[highest] == 0)
        {
            completed++;
            printf("P%d Completed at %d\n", highest + 1, time);
        }
    }

    return 0;
}
